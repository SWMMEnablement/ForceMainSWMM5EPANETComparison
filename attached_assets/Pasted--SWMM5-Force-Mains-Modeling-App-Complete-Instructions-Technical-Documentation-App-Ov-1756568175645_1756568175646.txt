# SWMM5 Force Mains Modeling App
## Complete Instructions & Technical Documentation

## 🚀 App Overview

This Replit application provides a comprehensive toolkit for modeling, analyzing, and optimizing force mains in SWMM5. The app helps engineers design pump station discharge systems, troubleshoot existing models, and ensure proper hydraulic performance of pressurized sewer systems.

### Key Features
- **Force Main Designer**: Interactive tool for sizing and configuring force mains
- **Friction Loss Calculator**: Compare Hazen-Williams vs Darcy-Weisbach methods
- **Network Analyzer**: Validate complex force main networks
- **Troubleshooting Assistant**: Diagnose and fix common modeling issues
- **SWMM5 Input File Generator**: Create properly formatted .inp files
- **Results Visualizer**: Plot hydraulic grade lines and system performance

```python
class ForceMaintCalculator:
    """
    Calculate hydraulic parameters for force mains
    """
    
    def __init__(self):
        self.gravity = 9.81  # m/s^2
        self.water_density = 998.2  # kg/m^3
        self.kinematic_viscosity = 1.004e-6  # m^2/s at 20°C
    
    def hazen_williams_loss(self, Q, D, L, C):
        """
        Calculate head loss using Hazen-Williams equation
        
        Parameters:
        Q: Flow rate (m³/s)
        D: Pipe diameter (m)
        L: Pipe length (m)
        C: Hazen-Williams coefficient (dimensionless)
        
        Returns:
        hf: Friction head loss (m)
        """
        # For SI units
        hf = 10.67 * (Q**1.852) * L / (C**1.852 * D**4.87)
        return hf
    
    def darcy_weisbach_loss(self, Q, D, L, e):
        """
        Calculate head loss using Darcy-Weisbach equation
        
        Parameters:
        Q: Flow rate (m³/s)
        D: Pipe diameter (m)
        L: Pipe length (m)
        e: Absolute roughness (m)
        
        Returns:
        hf: Friction head loss (m)
        """
        A = np.pi * D**2 / 4
        V = Q / A
        Re = V * D / self.kinematic_viscosity
        
        # Swamee-Jain approximation for friction factor
        f = 0.25 / (np.log10(e/(3.7*D) + 5.74/Re**0.9))**2
        
        hf = f * L * V**2 / (D * 2 * self.gravity)
        return hf
    
    def manning_to_hazen_williams(self, n, D, S):
        """
        Convert Manning's n to Hazen-Williams C
        """
        C = 1.067 / (n * (D / S)**0.04)
        return C
    
    def check_full_flow_condition(self, pump_head, static_head, friction_loss):
        """
        Verify if force main will flow full
        """
        available_head = pump_head - static_head
        if available_head > friction_loss:
            return True, available_head - friction_loss
        return False, available_head - friction_loss
```

### Module 2: Network Solver

```python
class ForceMainNetwork:
    """
    Solve complex force main networks using Newton-Raphson method
    """
    
    def __init__(self):
        self.nodes = {}
        self.links = {}
        self.pumps = {}
        self.tolerance = 0.001
        self.max_iterations = 100
    
    def add_wet_well(self, id, invert, max_depth, init_depth):
        """Add wet well to network"""
        self.nodes[id] = {
            'type': 'wet_well',
            'invert': invert,
            'max_depth': max_depth,
            'depth': init_depth,
            'head': invert + init_depth
        }
    
    def add_junction(self, id, invert, surcharge_depth=0):
        """Add junction chamber to network"""
        self.nodes[id] = {
            'type': 'junction',
            'invert': invert,
            'surcharge_depth': surcharge_depth,
            'depth': 0,
            'head': invert
        }
    
    def add_force_main(self, id, from_node, to_node, diameter, length, friction_params):
        """Add force main link"""
        self.links[id] = {
            'type': 'force_main',
            'from': from_node,
            'to': to_node,
            'diameter': diameter,
            'length': length,
            'friction': friction_params,
            'flow': 0
        }
    
    def solve_network(self):
        """
        Solve network using iterative Newton-Raphson method
        """
        iteration = 0
        converged = False
        
        while not converged and iteration < self.max_iterations:
            # Build system matrices
            A, b = self.build_matrices()
            
            # Solve linear system
            delta_H = np.linalg.solve(A, b)
            
            # Update heads
            self.update_heads(delta_H)
            
            # Check convergence
            converged = np.max(np.abs(delta_H)) < self.tolerance
            iteration += 1
        
        return converged, iteration
    
    def build_matrices(self):
        """Build Jacobian matrix and residual vector"""
        n = len(self.nodes)
        A = np.zeros((n, n))
        b = np.zeros(n)
        
        # Continuity equations for each node
        for i, node_id in enumerate(self.nodes):
            # Sum of flows = 0
            for link_id, link in self.links.items():
                if link['from'] == node_id:
                    # Outflow
                    A[i, :], b[i] = self.add_link_contribution(link, -1)
                elif link['to'] == node_id:
                    # Inflow
                    A[i, :], b[i] = self.add_link_contribution(link, 1)
        
        return A, b
```

```python
class SWMMInputGenerator:
    """
    Generate SWMM5 input files for force main systems
    """
    
    def __init__(self):
        self.title = "Force Main System"
        self.options = {
            'FLOW_UNITS': 'CMS',
            'INFILTRATION': 'HORTON',
            'FLOW_ROUTING': 'DYNWAVE',
            'FORCE_MAIN_EQUATION': 'H-W',
            'SURCHARGE_METHOD': 'SLOT',
            'VARIABLE_STEP': 0.75,
            'MIN_SURFAREA': 1.14,
            'NORMAL_FLOW_LIMITED': 'BOTH'
        }
        
    def create_force_main_system(self, config):
        """
        Create complete force main system
        
        config = {
            'wet_well': {...},
            'pump': {...},
            'force_main': {...},
            'discharge_node': {...}
        }
        """
        inp_content = []
        
        # Title section
        inp_content.append("[TITLE]")
        inp_content.append(f";;{self.title}")
        inp_content.append("")
        
        # Options section
        inp_content.append("[OPTIONS]")
        for key, value in self.options.items():
            inp_content.append(f"{key:<20} {value}")
        inp_content.append("")
        
        # Junctions section
        inp_content.append("[JUNCTIONS]")
        inp_content.append(";;Name  Elevation  MaxDepth  InitDepth  SurDepth  Aponded")
        
        # Add discharge node with surcharge
        discharge = config['discharge_node']
        inp_content.append(
            f"{discharge['id']:<12} {discharge['elevation']:8.2f} "
            f"{discharge['max_depth']:8.2f} {0:8.2f} "
            f"{discharge['surcharge_depth']:8.2f} {0:8.2f}"
        )
        inp_content.append("")
        
        # Storage section for wet well
        inp_content.append("[STORAGE]")
        inp_content.append(";;Name  Elev  MaxDepth  InitDepth  Shape  Curve/Params")
        
        ww = config['wet_well']
        inp_content.append(
            f"{ww['id']:<12} {ww['invert']:8.2f} "
            f"{ww['max_depth']:8.2f} {ww['init_depth']:8.2f} "
            f"FUNCTIONAL {ww['area']:8.2f} 0 0 0"
        )
        inp_content.append("")
        
        # Conduits section
        inp_content.append("[CONDUITS]")
        inp_content.append(";;Name  From  To  Length  Roughness  InOffset  OutOffset  InitFlow  MaxFlow")
        
        fm = config['force_main']
        inp_content.append(
            f"{fm['id']:<12} {fm['from']:<12} {fm['to']:<12} "
            f"{fm['length']:8.2f} {fm['roughness']:8.4f} "
            f"{0:8.2f} {0:8.2f} {0:8.2f} {0:8.2f}"
        )
        inp_content.append("")
        
        # XSections section
        inp_content.append("[XSECTIONS]")
        inp_content.append(";;Link  Shape  Geom1  Geom2  Geom3  Geom4  Barrels  Culvert")
        inp_content.append(
            f"{fm['id']:<12} FORCE_MAIN {fm['diameter']:8.3f} "
            f"0 0 0 1"
        )
        inp_content.append("")
        
        # Pumps section
        inp_content.append("[PUMPS]")
        inp_content.append(";;Name  From  To  Curve  Status  Startup  Shutoff")
        
        pump = config['pump']
        inp_content.append(
            f"{pump['id']:<12} {pump['from']:<12} {pump['to']:<12} "
            f"{pump['curve']:<12} ON {pump['startup']:8.2f} {pump['shutoff']:8.2f}"
        )
        inp_content.append("")
        
        # Curves section
        inp_content.append("[CURVES]")
        inp_content.append(f";;Name  Type  X-Value  Y-Value")
        
        for point in pump['curve_data']:
            inp_content.append(
                f"{pump['curve']:<12} Pump4 {point[0]:8.3f} {point[1]:8.2f}"
            )
        
        return "\n".join(inp_content)
```

```python
class TroubleshootingAssistant:
    """
    Diagnose and fix common force main modeling issues
    """
    
    def __init__(self):
        self.issues = []
        self.recommendations = []
    
    def check_force_main_full(self, model_results):
        """
        Check if force main maintains full flow
        """
        for time, data in model_results.items():
            d_over_D = data['depth'] / data['diameter']
            
            if d_over_D < 0.95:
                self.issues.append({
                    'severity': 'HIGH',
                    'issue': f'Force main not full at time {time}',
                    'd/D': d_over_D,
                    'location': data['link_id']
                })
                
                # Generate recommendations
                self.add_recommendations_for_partial_flow()
    
    def add_recommendations_for_partial_flow(self):
        """
        Provide solutions for partial flow issues
        """
        self.recommendations.extend([
            {
                'priority': 1,
                'action': 'Add Break Node',
                'details': '''
                1. Insert a manhole at the force main outlet
                2. Set surcharge depth = 1.5 * pipe diameter
                3. This maintains backpressure to keep pipe full
                ''',
                'code_example': '''
                [JUNCTIONS]
                BreakNode  100.0  3.0  0  1.5  0
                '''
            },
            {
                'priority': 2,
                'action': 'Install Flap Gate',
                'details': '''
                1. Add flap gate to prevent backflow
                2. Set type = FLAP in [OUTLETS] section
                3. Maintains unidirectional flow
                ''',
                'code_example': '''
                [OUTLETS]
                FlapGate  Node1  Node2  TABULAR/HEAD  FlowCurve
                '''
            },
            {
                'priority': 3,
                'action': 'Increase Downstream Roughness',
                'details': '''
                1. Multiply gravity main roughness by 2-3x
                2. Simulates transition losses
                3. Maintains higher upstream depth
                ''',
                'calculation': '''
                n_modified = n_original * 2.5
                '''
            },
            {
                'priority': 4,
                'action': 'Reduce Time Step',
                'details': '''
                1. Set routing time step ≤ 5 seconds
                2. Enable variable time step
                3. Use safety factor = 0.75
                ''',
                'settings': '''
                [OPTIONS]
                ROUTING_STEP        5
                VARIABLE_STEP       0.75
                '''
            }
        ])
    
    def check_pump_cycling(self, pump_data, time_step):
        """
        Detect rapid pump cycling issues
        """
        on_off_events = []
        
        for i in range(1, len(pump_data)):
            if pump_data[i]['status'] != pump_data[i-1]['status']:
                on_off_events.append(pump_data[i]['time'])
        
        if len(on_off_events) > 1:
            min_cycle_time = min(np.diff(on_off_events))
            
            if min_cycle_time < 10 * time_step:
                self.issues.append({
                    'severity': 'MEDIUM',
                    'issue': 'Rapid pump cycling detected',
                    'min_cycle_time': min_cycle_time,
                    'recommended_timestep': min_cycle_time / 10
                })
    
    def validate_friction_coefficients(self, pipe_material, coefficient, method):
        """
        Validate friction coefficients are reasonable
        """
        valid_ranges = {
            'Hazen-Williams': {
                'PVC': (130, 150),
                'Ductile Iron': (100, 140),
                'Concrete': (100, 130),
                'Steel': (90, 120)
            },
            'Darcy-Weisbach': {
                'PVC': (0.0015, 0.003),
                'Ductile Iron': (0.025, 0.12),
                'Concrete': (0.3, 3.0),
                'Steel': (0.045, 0.09)
            }
        }
        
        if method in valid_ranges:
            if pipe_material in valid_ranges[method]:
                min_val, max_val = valid_ranges[method][pipe_material]
                if not (min_val <= coefficient <= max_val):
                    self.issues.append({
                        'severity': 'LOW',
                        'issue': f'Unusual {method} coefficient',
                        'value': coefficient,
                        'expected_range': (min_val, max_val),
                        'material': pipe_material
                    })
```

```python
class ForceMainVisualizer:
    """
    Create interactive visualizations for force main analysis
    """
    
    def __init__(self):
        self.figures = []
    
    def plot_hydraulic_grade_line(self, profile_data):
        """
        Plot HGL along force main alignment
        """
        import plotly.graph_objects as go
        
        fig = go.Figure()
        
        # Ground profile
        fig.add_trace(go.Scatter(
            x=profile_data['distance'],
            y=profile_data['ground_elevation'],
            mode='lines',
            name='Ground Surface',
            line=dict(color='brown', width=2)
        ))
        
        # Pipe profile
        fig.add_trace(go.Scatter(
            x=profile_data['distance'],
            y=profile_data['pipe_invert'],
            mode='lines',
            name='Pipe Invert',
            line=dict(color='black', width=3)
        ))
        
        fig.add_trace(go.Scatter(
            x=profile_data['distance'],
            y=profile_data['pipe_crown'],
            mode='lines',
            name='Pipe Crown',
            line=dict(color='black', width=3, dash='dash')
        ))
        
        # HGL
        fig.add_trace(go.Scatter(
            x=profile_data['distance'],
            y=profile_data['hgl'],
            mode='lines',
            name='Hydraulic Grade Line',
            line=dict(color='blue', width=2)
        ))
        
        # Add annotations for key points
        fig.add_annotation(
            x=0, y=profile_data['hgl'][0],
            text="Pump Station",
            showarrow=True,
            arrowhead=2
        )
        
        fig.update_layout(
            title="Force Main Hydraulic Profile",
            xaxis_title="Distance (m)",
            yaxis_title="Elevation (m)",
            hovermode='x unified'
        )
        
        return fig
    
    def plot_pump_curve_operation(self, pump_curve, operating_points):
        """
        Show pump curve with operating points
        """
        import matplotlib.pyplot as plt
        
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
        
        # Head-Flow curve
        ax1.plot(pump_curve['flow'], pump_curve['head'], 
                'b-', linewidth=2, label='Pump Curve')
        ax1.scatter(operating_points['flow'], operating_points['head'],
                   c='red', s=50, label='Operating Points')
        ax1.set_xlabel('Flow (m³/s)')
        ax1.set_ylabel('Head (m)')
        ax1.set_title('Pump Performance Curve')
        ax1.grid(True, alpha=0.3)
        ax1.legend()
        
        # Efficiency curve
        ax2.plot(pump_curve['flow'], pump_curve['efficiency'],
                'g-', linewidth=2)
        ax2.scatter(operating_points['flow'], operating_points['efficiency'],
                   c='red', s=50)
        ax2.set_xlabel('Flow (m³/s)')
        ax2.set_ylabel('Efficiency (%)')
        ax2.set_title('Pump Efficiency')
        ax2.grid(True, alpha=0.3)
        
        plt.tight_layout()
        return fig
    
    def create_network_diagram(self, network):
        """
        Generate interactive network diagram
        """
        import networkx as nx
        import plotly.graph_objects as go
        
        # Create graph
        G = nx.DiGraph()
        
        # Add nodes
        for node_id, node_data in network.nodes.items():
            G.add_node(node_id, **node_data)
        
        # Add edges (pipes and pumps)
        for link_id, link_data in network.links.items():
            G.add_edge(link_data['from'], link_data['to'],
                      id=link_id, **link_data)
        
        # Generate layout
        pos = nx.spring_layout(G)
        
        # Create Plotly figure
        edge_trace = []
        for edge in G.edges():
            x0, y0 = pos[edge[0]]
            x1, y1 = pos[edge[1]]
            edge_trace.append(go.Scatter(
                x=[x0, x1, None],
                y=[y0, y1, None],
                mode='lines',
                line=dict(width=2, color='gray'),
                hoverinfo='none'
            ))
        
        node_trace = go.Scatter(
            x=[pos[node][0] for node in G.nodes()],
            y=[pos[node][1] for node in G.nodes()],
            mode='markers+text',
            text=[str(node) for node in G.nodes()],
            textposition="top center",
            marker=dict(
                size=15,
                color=['blue' if G.nodes[node]['type'] == 'wet_well' 
                       else 'green' for node in G.nodes()],
                line=dict(width=2, color='black')
            ),
            hoverinfo='text'
        )
        
        fig = go.Figure(data=edge_trace + [node_trace])
        fig.update_layout(
            title="Force Main Network Diagram",
            showlegend=False,
            hovermode='closest',
            xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
            yaxis=dict(showgrid=False, zeroline=False, showticklabels=False)
        )
        
        return fig
```

```python
# api.py - REST API for external integration
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

app = FastAPI()

class ForceMainRequest(BaseModel):
    flow: float
    diameter: float
    length: float
    method: str
    coefficient: float

@app.post("/calculate")
async def calculate_force_main(request: ForceMainRequest):
    calc = ForceMainCalculator()
    
    if request.method == "HW":
        loss = calc.hazen_williams_loss(
            request.flow, request.diameter, 
            request.length, request.coefficient
        )
    else:
        loss = calc.darcy_weisbach_loss(
            request.flow, request.diameter,
            request.length, request.coefficient
        )
    
    return {"head_loss": loss}
```
